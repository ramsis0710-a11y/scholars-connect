<#
  DEPLOIEMENT DE answer-enricher.js (v16.22) - scholars-connect (GitHub -> Render)

  Ce script contient le fichier complet (compresse). Il :
    1. verifie Git / Node et le depot local,
    2. met le depot a jour (git pull),
    3. verifie l integrite du fichier (SHA-256) et sa syntaxe (node --check),
    4. sauvegarde l ancien answer-enricher.js sur le Bureau,
    5. remplace le fichier, verifie pdf-parse dans package.json,
    6. commit puis push vers GitHub (Render redeploie automatiquement).

  Utilisation (Windows PowerShell) :
    powershell -ExecutionPolicy Bypass -File "$env:USERPROFILE\Downloads\deploy-answer-enricher.ps1"

  Options :
    -RepoPath  dossier du depot local (defaut : C:\Users\PC\Desktop\scholars-connect-clean)
    -Yes       pas de question avant le push
    -NoPush    commit local uniquement, sans push
#>
param(
  [string]$RepoPath = 'C:\Users\PC\Desktop\scholars-connect-clean',
  [string]$ExpectedRemote = 'ramsis0710-a11y/scholars-connect',
  [string]$Branch = 'main',
  [switch]$Yes,
  [switch]$NoPush
)

$ErrorActionPreference = 'Continue'
$TargetFile = 'answer-enricher.js'
$NewVersion = 'v16.22'
$ExpectedSha256 = '1116e54aed1b13d2d108d03246ba33cce647bf7df093268bb4db847fb1b1bd37'
$ExpectedBytes = 226303

function Say([string]$text, [string]$color = 'Gray') { Write-Host $text -ForegroundColor $color }
function Stop-Deploy([string]$text) {
  Write-Host ''
  Write-Host ('ERREUR : ' + $text) -ForegroundColor Red
  exit 1
}

Say '=== Deploiement de answer-enricher.js ===' 'Cyan'

# ---------- 1. Outils ----------
Say '[1/7] Verification des outils...'
if (-not (Get-Command git -ErrorAction SilentlyContinue)) {
  Stop-Deploy 'Git est introuvable. Installez Git for Windows : https://git-scm.com/download/win'
}
$hasNode = [bool](Get-Command node -ErrorAction SilentlyContinue)
if (-not $hasNode) { Say '    Node.js introuvable : la verification de syntaxe sera ignoree.' 'Yellow' }

# ---------- 2. Depot local ----------
if (-not (Test-Path -LiteralPath $RepoPath)) { Stop-Deploy ('Dossier introuvable : ' + $RepoPath) }
Set-Location -LiteralPath $RepoPath
if (-not (Test-Path -LiteralPath (Join-Path $RepoPath '.git'))) {
  Stop-Deploy ('Ce dossier n est pas un depot Git (pas de dossier .git) : ' + $RepoPath)
}
$remote = ((& git remote get-url origin 2>$null) | Out-String).Trim()
if ($remote -notlike ('*' + $ExpectedRemote + '*')) {
  Stop-Deploy ('Le depot pointe vers "' + $remote + '" et non vers ' + $ExpectedRemote)
}
$current = ((& git rev-parse --abbrev-ref HEAD 2>$null) | Out-String).Trim()
if ($current -ne $Branch) {
  Stop-Deploy ('Branche actuelle : "' + $current + '". Passez sur ' + $Branch + ' avec : git checkout ' + $Branch)
}
$gitUser = ((& git config user.name 2>$null) | Out-String).Trim()
$gitMail = ((& git config user.email 2>$null) | Out-String).Trim()
if (-not $gitUser -or -not $gitMail) {
  Stop-Deploy 'Identite Git absente. Executez : git config --global user.name "Votre nom"  puis  git config --global user.email "vous@exemple.com"'
}
Say ('    Depot : ' + $RepoPath + '  (' + $remote + ', branche ' + $current + ')') 'Green'

Say '[2/7] Mise a jour du depot (git pull)...'
$dirty = ((& git status --porcelain 2>$null) | Out-String).Trim()
if ($dirty) {
  Say '    Modifications locales presentes (seul answer-enricher.js sera commite) :' 'Yellow'
  Say $dirty 'DarkYellow'
}
& git pull --ff-only origin $Branch
if ($LASTEXITCODE -ne 0) {
  Stop-Deploy 'git pull a echoue. Resolvez la situation du depot (git status), puis relancez le script.'
}

try {
  $gz = [Convert]::FromBase64String(($PayloadB64 -replace '\s', ''))
  $msIn = New-Object System.IO.MemoryStream -ArgumentList (, $gz)
  $gs = New-Object System.IO.Compression.GzipStream -ArgumentList $msIn, ([System.IO.Compression.CompressionMode]::Decompress)
  $msOut = New-Object System.IO.MemoryStream
  $gs.CopyTo($msOut)
  $gs.Dispose()
  $newBytes = $msOut.ToArray()
} catch {
  Stop-Deploy ('Decodage du contenu impossible : ' + $_.Exception.Message)
}
$sha = [System.Security.Cryptography.SHA256]::Create()
$newHash = ([BitConverter]::ToString($sha.ComputeHash($newBytes)) -replace '-', '').ToLower()
if ($newBytes.Length -ne $ExpectedBytes -or $newHash -ne $ExpectedSha256) {
  Stop-Deploy ('Integrite du fichier incorrecte (script modifie ou tronque). Attendu ' + $ExpectedBytes + ' octets / ' + $ExpectedSha256 + ' ; obtenu ' + $newBytes.Length + ' octets / ' + $newHash)
}
Say ('    Integrite verifiee : ' + $newBytes.Length + ' octets, SHA-256 ' + $newHash.Substring(0, 16) + '...') 'Green'

$tmp = Join-Path ([System.IO.Path]::GetTempPath()) ('answer-enricher.' + $NewVersion + '.check.js')
[System.IO.File]::WriteAllBytes($tmp, $newBytes)
if ($hasNode) {
  & node --check $tmp
  if ($LASTEXITCODE -ne 0) { Stop-Deploy 'La verification de syntaxe JavaScript (node --check) a echoue.' }
  Say '    Syntaxe JavaScript valide (node --check).' 'Green'
}

# ---------- 4. Sauvegarde de l ancien fichier ----------
Say '[4/7] Sauvegarde de l ancien fichier...'
$target = Join-Path $RepoPath $TargetFile
$oldVersion = 'absent'
if (Test-Path -LiteralPath $target) {
  $oldHash = ([BitConverter]::ToString($sha.ComputeHash([System.IO.File]::ReadAllBytes($target))) -replace '-', '').ToLower()
  if ($oldHash -eq $newHash) {
    Say ('    Le depot contient deja la version ' + $NewVersion + ' : rien a faire.') 'Green'
    Remove-Item -LiteralPath $tmp -ErrorAction SilentlyContinue
    exit 0
  }
  $oldVersion = 'inconnue'
  $m = Select-String -LiteralPath $target -Pattern "APP_VERSION\s*=\s*'([^']+)'" | Select-Object -First 1
  if ($m) { $oldVersion = $m.Matches[0].Groups[1].Value }
  $desktop = [Environment]::GetFolderPath('Desktop')
  if (-not $desktop) { $desktop = $HOME }
  $backupDir = Join-Path $desktop 'scholars-connect-backups'
  New-Item -ItemType Directory -Force -Path $backupDir | Out-Null
  $stamp = Get-Date -Format 'yyyyMMdd-HHmmss'
  $backup = Join-Path $backupDir ('answer-enricher.' + $oldVersion + '.' + $stamp + '.js')
  Copy-Item -LiteralPath $target -Destination $backup -Force
  Say ('    Ancien fichier (' + $oldVersion + ') sauvegarde : ' + $backup) 'Green'
} else {
  Say '    Aucun answer-enricher.js dans le depot : creation.' 'Yellow'
}

# ---------- 5. Remplacement + dependances ----------
Say '[5/7] Remplacement du fichier et verification de pdf-parse...'
[System.IO.File]::WriteAllBytes($target, $newBytes)
Remove-Item -LiteralPath $tmp -ErrorAction SilentlyContinue
$paths = @($TargetFile)
$pkg = Join-Path $RepoPath 'package.json'
if (Test-Path -LiteralPath $pkg) {
  $pkgText = [System.IO.File]::ReadAllText($pkg)
  if ($pkgText -notmatch '"pdf-parse"') {
    if (Get-Command npm -ErrorAction SilentlyContinue) {
      Say '    pdf-parse absent de package.json : installation de la version 1.1.1...' 'Yellow'
      & npm install pdf-parse@1.1.1 --save-exact --no-audit --no-fund
      if ($LASTEXITCODE -ne 0) { Stop-Deploy 'npm install pdf-parse@1.1.1 a echoue.' }
      $paths += 'package.json'
      if (Test-Path -LiteralPath (Join-Path $RepoPath 'package-lock.json')) { $paths += 'package-lock.json' }
    } else {
      Say '    npm introuvable : ajoutez "pdf-parse": "1.1.1" dans les dependencies de package.json.' 'Yellow'
    }
  } else {
    Say '    pdf-parse deja present dans package.json.' 'Green'
  }
} else {
  Say '    package.json introuvable : verification de pdf-parse ignoree.' 'Yellow'
}

# ---------- 6. Commit ----------
Say '[6/7] Commit...'
& git --no-pager diff --stat -- $TargetFile
Say ('    ' + $oldVersion + '  ->  ' + $NewVersion) 'Cyan'
& git add -- $paths

$msg = 'answer-enricher ' + $NewVersion + ' : fusion Version Finale v16.21.1 + extraction ARC v16.8 (lignes avec prix, multi-ARC, lignes de service, HYD 511)'
