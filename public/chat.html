function cleanForSpeech(text) { if (!text) return ''; let t = String(text); t = t.replace(/\*\*/g, ''); t = t.replace(/\*/g, ''); t = t.replace(/^#+\s*/gm, ''); t = t.replace(/_/g, ' '); t = t.replace(/~/g, ' '); t = t.replace(/\|/g, ' '); t = t.replace(/->/g, ' vers '); t = t.replace(/=>/g, ' donne '); t = t.replace(/<=/g, ' inferieur ou egal a '); t = t.replace(/>=/g, ' superieur ou egal a '); t = t.replace(/=/g, ' egale '); t = t.replace(/\+/g, ' plus '); t = t.replace(/(\d)\s*-\s*(\d)/g, ' moins $2'); t = t.replace(/(\d)\s*\*\s*(\d)/g, ' fois $2'); t = t.replace(/(\d)\s*x\s*(\d)/g, ' fois $2'); t = t.replace(/(\d)\s*\/\s*(\d)/g, ' divise par $2'); t = t.replace(/²/g, ' au carre '); t = t.replace(/³/g, ' au cube '); t = t.replace(/√/g, ' racine carree de '); t = t.replace(/π/g, ' pi '); t = t.replace(/°/g, ' degres '); t = t.replace(/%/g, ' pour cent '); t = t.replace(/^[-•·]\s*/gm, ''); t = t.replace(/^\d+\.\s*/gm, ''); t = t.replace(/\s+/g, ' '); return t.trim(); } function speak(text, lang = 'fr-FR') { if (!('speechSynthesis' in window)) { alert('Synthese vocale non supportee.'); return; } const cleanText = cleanForSpeech(text); if (!cleanText) return; speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(cleanText); u.lang = lang; u.rate = 0.88; u.pitch = 0.85; u.volume = 1.0; const voices = speechSynthesis.getVoices(); const langPrefix = lang.split('-')[0]; const preferred = voices.find(v => v.lang.startsWith(langPrefix) && /Thomas|Henri|Paul|Guillaume|Yannick/i.test(v.name)); const maleAny = voices.find(v => v.lang.startsWith(langPrefix) && /male|homme/i.test(v.name)); const langVoice = voices.find(v => v.lang.startsWith(langPrefix)); if (preferred) u.voice = preferred; else if (maleAny) u.voice = maleAny; else if (langVoice) u.voice = langVoice; speechSynthesis.speak(u); }
